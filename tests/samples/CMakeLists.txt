# Microsoft's public samples built as Gaming.Xbox.Scarlett.x64 executables with the
# GDKX defines and import libraries (Platform.Common.props / Platform.Edition.props), so the
# layer runs genuine Series X|S binaries. Each sample's own .vcxproj supplies its sources,
# shaders and assets (tools/samples/vcxproj2cmake.py). Outputs link SDK import libraries and
# stay in the build tree (hygiene rule 2); only these scripts are versioned.
set(XSL_WINE "wine" CACHE STRING "Wine binary used to run SDK tools at build time")
set(XSL_TOOL_PREFIX "${CMAKE_BINARY_DIR}/wineprefix-tools" CACHE PATH "Wine prefix for build-time SDK tools")
set(XSL_DXC "${GXDK_ROOT}/xbox/bin/gen9/dxc.exe")
set(XSL_SAMPLE_SHIM "${CMAKE_CURRENT_SOURCE_DIR}/shim")
set(XSL_SAMPLE_KITS "${XSL_SAMPLES_ROOT}/Kits")
set(XSL_TOOLS "${CMAKE_SOURCE_DIR}/tools/samples")

# Header names the samples spell with different case than the SDK ships (Linux is case-sensitive).
# XSL_CASE_DIR (case-alias include links) is generated by the top-level CMakeLists.txt.

# Common Scarlett compile/link settings, as the GDKX MSBuild platform files set them.
function(xsl_scarlett_settings tgt)
  set_target_properties(${tgt} PROPERTIES MSVC_RUNTIME_LIBRARY "MultiThreadedDLL")
  target_compile_definitions(${tgt} PRIVATE _GAMING_XBOX _GAMING_XBOX_SCARLETT WINAPI_FAMILY=WINAPI_FAMILY_GAMES
    __WRL_NO_DEFAULT_LIB__ _ATL_NO_DEFAULT_LIBS NDEBUG)
  target_compile_options(${tgt} PRIVATE /arch:AVX2 /Zc:char8_t- /O2 -Wno-everything "/FI${XSL_SAMPLE_SHIM}/xsl_sample_shim.h")
  target_include_directories(${tgt} BEFORE PRIVATE "${XSL_SAMPLE_SHIM}" "${XSL_CASE_DIR}"
    "${GXDK_ROOT}/xbox/include/gen9" "${GXDK_ROOT}/xbox/include")
endfunction()

# ---- Samples ------------------------------------------------------------------------------
# xsl_add_sample(<name> <path to .vcxproj> [<extra define>...])
function(xsl_add_sample name vcxproj)
  set(meta "${CMAKE_CURRENT_BINARY_DIR}/${name}.vcx.cmake")
  execute_process(COMMAND ${Python3_EXECUTABLE} "${XSL_TOOLS}/vcxproj2cmake.py" "${vcxproj}" "${meta}" RESULT_VARIABLE rc)
  if(rc)
    message(WARNING "skipping ${name}: cannot read ${vcxproj}")
    return()
  endif()
  set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${vcxproj}")
  include("${meta}")
  set(outdir "${CMAKE_BINARY_DIR}/samples/${name}")
  add_executable(${name} WIN32 ${S_SOURCES})
  set_target_properties(${name} PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${outdir}")
  xsl_scarlett_settings(${name})
  target_compile_definitions(${name} PRIVATE ${S_DEFINES} ${ARGN})
  target_include_directories(${name} PRIVATE ${S_INCLUDES} "${DXTK}/Inc")
  target_link_directories(${name} PRIVATE "${GXDK_ROOT}/xbox/lib/gen9" "${GXDK_ROOT}/xbox/lib/x64")
  target_link_options(${name} PRIVATE /nodefaultlib:kernel32.lib /nodefaultlib:user32.lib /nodefaultlib:ole32.lib)
  target_link_libraries(${name} PRIVATE dxtk12_scarlett
    PIXEvt.lib d3d12_xs.lib GameInput.lib xgameplatform.lib xgameruntime.lib xmem.lib xg_xs.lib)
  # Further SDK import libraries the project names (e.g. dstorage_xs.lib), if this SDK has them.
  foreach(lib ${S_EXTRA_LIBS})
    if(EXISTS "${GXDK_ROOT}/xbox/lib/gen9/${lib}" OR EXISTS "${GXDK_ROOT}/xbox/lib/x64/${lib}")
      target_link_libraries(${name} PRIVATE ${lib})
    endif()
  endforeach()
  # vcpkg dependencies the layer vendors.
  if("zlib" IN_LIST S_VCPKG)
    target_link_libraries(${name} PRIVATE xsl_zlib)
  endif()

  set(outs)
  foreach(sh ${S_SHADERS})
    string(REPLACE "|" ";" p "${sh}")
    list(GET p 0 file)
    list(GET p 1 profile)
    list(GET p 2 entry)
    list(GET p 3 base)
    list(GET p 4 opts)
    string(REPLACE "@OUTDIR@" "Z:${outdir}/" opts "${opts}")
    separate_arguments(opts WINDOWS_COMMAND "${opts}")
    get_filename_component(dir "${file}" DIRECTORY)
    set(out "${outdir}/${base}.cso")
    if(profile STREQUAL "rootsig_1_1")
      set(eflag -E ${entry})
    else()
      set(eflag -E ${entry})
    endif()
    get_filename_component(fname "${file}" NAME)
    # dxc runs under Wine: give it Windows-style (Z:) paths and run it from the shader's directory.
    # Forward slashes: ninja runs commands through sh, which would eat backslashes.
    set(kits_w "Z:${XSL_SAMPLE_KITS}")
    set(sh_incs)
    foreach(inc ${S_SHADER_INCLUDES})
      list(APPEND sh_incs -I "Z:${inc}")
    endforeach()
    set(out_w "Z:${out}")
    add_custom_command(OUTPUT "${out}"
      COMMAND ${CMAKE_COMMAND} -E make_directory "${outdir}"
      COMMAND ${CMAKE_COMMAND} -E env WINEPREFIX=${XSL_TOOL_PREFIX} WINEDEBUG=-all
              ${XSL_WINE} "${XSL_DXC}" -nologo -T ${profile} ${eflag} ${opts}
              -I . ${sh_incs} -I "${kits_w}/ATGTK" -I "${kits_w}/ATGTK/HDR" "${fname}" -Fo "${out_w}"
      WORKING_DIRECTORY "${dir}"
      DEPENDS "${file}" COMMENT "dxc_xs ${name}/${base}")
    list(APPEND outs "${out}")
  endforeach()
  foreach(a ${S_ASSETS})
    string(REPLACE "|" ";" p "${a}")
    list(GET p 0 src)
    list(LENGTH p n)
    set(sub "")
    set(dname "")
    if(n GREATER 1)
      list(GET p 1 sub)
    endif()
    if(n GREATER 2)
      list(GET p 2 dname)
    endif()
    if(NOT EXISTS "${src}")
      message(STATUS "${name}: asset not in checkout, skipped: ${src}")
      continue()
    endif()
    get_filename_component(fname "${src}" NAME)
    if(dname)
      set(fname "${dname}")
    endif()
    set(dst "${outdir}/${sub}/${fname}")
    add_custom_command(OUTPUT "${dst}" COMMAND ${CMAKE_COMMAND} -E copy_if_different "${src}" "${dst}" DEPENDS "${src}")
    list(APPEND outs "${dst}")
  endforeach()
  if(outs)
    add_custom_target(${name}_content ALL DEPENDS ${outs})
    add_dependencies(${name} ${name}_content)
  endif()
endfunction()

# Samples rebuilt with D3DCOMPILE_NO_DEBUG, as retail titles are, so d3d12_xs.h inlines
# CreateConstantBufferView (Xbox-format descriptors written straight into descriptor memory) and
# CopyDescriptorsSimple (raw descriptor copies). Only the sample's own code takes the define; the

  IntroGraphics/SimpleBezier Graphics/ExecuteIndirect

# ---- Synthetic Series X|S titles (tests/synthetic) -----------------------------------------------
# xsl_add_synthetic(<name> <source> [<compute shader .hlsl> ...]): shaders become <stem>.cso next to the exe.
function(xsl_add_synthetic name src)
  set(outdir "${CMAKE_BINARY_DIR}/synthetic/${name}")
  add_executable(${name} "${src}")
  set_target_properties(${name} PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${outdir}")
  set(outs)
  foreach(hlsl ${ARGN})
    get_filename_component(stem "${hlsl}" NAME_WE)
    get_filename_component(dir "${hlsl}" DIRECTORY)
    get_filename_component(fname "${hlsl}" NAME)
    set(out "${outdir}/${stem}.cso")
    add_custom_command(OUTPUT "${out}"
      COMMAND ${CMAKE_COMMAND} -E make_directory "${outdir}"
      COMMAND ${CMAKE_COMMAND} -E env WINEPREFIX=${XSL_TOOL_PREFIX} WINEDEBUG=-all
              ${XSL_WINE} "${XSL_DXC}" -nologo -T cs_6_0 -E main "${fname}" -Fo "Z:${out}"
      WORKING_DIRECTORY "${dir}" DEPENDS "${hlsl}" COMMENT "dxc_xs ${name}/${stem}")
    list(APPEND outs "${out}")
  endforeach()
  if(outs)
    add_custom_target(${name}_content ALL DEPENDS ${outs})
    add_dependencies(${name} ${name}_content)
  endif()
  xsl_scarlett_settings(${name})
  target_link_directories(${name} PRIVATE "${GXDK_ROOT}/xbox/lib/gen9" "${GXDK_ROOT}/xbox/lib/x64")
  target_link_options(${name} PRIVATE /nodefaultlib:kernel32.lib /nodefaultlib:user32.lib /nodefaultlib:ole32.lib)
  target_link_libraries(${name} PRIVATE PIXEvt.lib d3d12_xs.lib GameInput.lib xgameplatform.lib xgameruntime.lib xmem.lib xg_xs.lib)
endfunction()

# HlslCompile's three PixelShader_DxCompiler_* shaders come from its own host tool, MyHlslCompiler,
# which calls the SDK compiler's API (dxcompiler_xs.dll, devpctools). Build it as a PC console
# tool and run it under Wine, as the project's custom build steps do.
set(XSL_HLSLCOMPILE "${XSL_SAMPLES_ROOT}/Samples/Graphics/HlslCompile")
if(TARGET HlslCompile AND EXISTS "${XSL_HLSLCOMPILE}/MyHlslCompiler/MyHlslCompiler.cpp")
  set(tooldir "${CMAKE_BINARY_DIR}/tools/MyHlslCompiler")
  add_executable(MyHlslCompiler "${XSL_HLSLCOMPILE}/MyHlslCompiler/MyHlslCompiler.cpp")
  set_target_properties(MyHlslCompiler PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${tooldir}" MSVC_RUNTIME_LIBRARY "MultiThreaded")
  target_compile_definitions(MyHlslCompiler PRIVATE _TOOL_XBOX_SCARLETT _TOOL_XBOX _CONSOLE UNICODE _UNICODE)
  target_compile_options(MyHlslCompiler PRIVATE -Wno-everything)
  target_include_directories(MyHlslCompiler PRIVATE "${GXDK_ROOT}/xbox/include/devpctools/gen9" "${GXDK_ROOT}/xbox/include/devpctools")
  target_link_directories(MyHlslCompiler PRIVATE "${GXDK_ROOT}/xbox/lib/devpctools/gen9")
  target_link_libraries(MyHlslCompiler PRIVATE shlwapi)
  # The compiler and the back ends it loads from its own directory (the set dxc.exe runs with).
  foreach(dll dxcompiler_xs.dll newbe_xs.dll xbsc_xs.dll PhSO_pc_xs.dll xgs12_pc_xs.dll XBTC.dll)
    add_custom_command(TARGET MyHlslCompiler POST_BUILD
      COMMAND ${CMAKE_COMMAND} -E copy_if_different "${GXDK_ROOT}/xbox/bin/gen9/${dll}" "${tooldir}/")
  endforeach()
  set(outdir "${CMAKE_BINARY_DIR}/samples/HlslCompile")
  set(outs)
  foreach(spec "EmbeddedPdb|0|" "AutoPdb|3|" "ManualPdb|3|pdb")
    string(REPLACE "|" ";" p "${spec}")
    list(GET p 0 kind)
    list(GET p 1 flags)
    list(GET p 2 pdb)
    set(stem "PixelShader_DxCompiler_${kind}")
    set(out "${outdir}/${stem}.cso")
    set(extra)
    if(pdb)
      set(extra "Z:${outdir}/${stem}.pdb")
    endif()
    add_custom_command(OUTPUT "${out}"
      COMMAND ${CMAKE_COMMAND} -E make_directory "${outdir}"
      COMMAND ${CMAKE_COMMAND} -E env WINEPREFIX=${XSL_TOOL_PREFIX} WINEDEBUG=-all "WINEDLLOVERRIDES=winedbg.exe=d"
              ${XSL_WINE} "${tooldir}/MyHlslCompiler.exe" dxc ${flags} "${stem}.hlsl" "Z:${out}" ${extra}
      WORKING_DIRECTORY "${XSL_HLSLCOMPILE}"
      DEPENDS MyHlslCompiler "${XSL_HLSLCOMPILE}/${stem}.hlsl" COMMENT "MyHlslCompiler ${stem}")
    list(APPEND outs "${out}")
  endforeach()
  add_custom_target(HlslCompile_dxcompiler ALL DEPENDS ${outs})
  add_dependencies(HlslCompile HlslCompile_dxcompiler)
endif()
